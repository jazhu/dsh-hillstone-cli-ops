  node %USERPROFILE%\.dsh\skills\dsh-panel-screenshots\scripts\shots.mjs \
    --config %USERPROFILE%\.dsh\skills\dsh-panel-screenshots\scripts\shots.config.mjs \
    --out docs/assets
